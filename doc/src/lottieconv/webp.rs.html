6auto_vectorize! {
7	pub(crate) fn bgra_to_rgba(buf_bgra: &[BGRA8], buf_rgba: &mut [RGBA8]) {
8		for i in 0..buf_bgra.len() {
9			buf_rgba[i].r = buf_bgra[i].r;
10			buf_rgba[i].g = buf_bgra[i].g;
11			buf_rgba[i].b = buf_bgra[i].b;
12			buf_rgba[i].a = buf_bgra[i].a;
13		}
14	}
15}

19pub struct Convert2Webp {

25impl Convert2Webp {

32	}

35impl Convert for Convert2Webp {
36	type Out = WebPData;

38
39	fn convert_frame(&self, from: &[BGRA8], to: &mut [RGBA8]) {
40		bgra_to_rgba(from, to)
41	}
42
43	fn add_frame(&mut self, data: &mut [u8]) -> Result<()> {
44		self.encoder.add_frame(data, self.timestamp.round() as _)?;
45		self.timestamp += self.delay;
46		Ok(())
47	}
48
49	fn finish(self) -> Result<Self::Out> {
50		self.encoder.finalize(self.timestamp.round() as _)
